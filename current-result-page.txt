"use client";

import {
  ArrowLeft,
  ArrowRight,
  BarChart3,
  CheckCircle2,
  CircleDollarSign,
  Clock3,
  Layers3,
  ShieldCheck,
  Sparkles,
  Trophy,
  TrendingDown,
} from "lucide-react";
import Link from "next/link";
import type { ReactNode } from "react";
import { useParams } from "next/navigation";
import { useCallback, useEffect, useMemo, useState } from "react";

import {
  getAllDeals,
  getDeal,
  type DealSession,
} from "@/lib/dealer-api";

type DealResponse = Awaited<ReturnType<typeof getDeal>>;
type Offer = NonNullable<DealResponse["offers"]>[number];
type Audit = NonNullable<DealResponse["audit"]>[number];

const money = (value: number) =>
  `â‚¹${Math.round(value).toLocaleString("en-IN")}`;

export default function DealResultPage() {
  const params = useParams<{ dealId: string }>();
  const dealId = params?.dealId;

  const [session, setSession] = useState<DealSession | null>(null);
  const [offers, setOffers] = useState<Offer[]>([]);
  const [audit, setAudit] = useState<Audit[]>([]);
  const [history, setHistory] = useState<DealSession[]>([]);
  const [error, setError] = useState("");

  const load = useCallback(async () => {
    if (!dealId) return;

    try {
      setError("");

      const current = await getDeal(dealId);

      if (!current.success || !current.session) {
        throw new Error(current.error ?? "Unable to load final deal");
      }

      setSession(current.session);
      setOffers(current.offers ?? []);
      setAudit(current.audit ?? []);

      const all = await getAllDeals();

      if (all.success) {
        setHistory(
          (all.deals ?? [])
            .filter(
              (deal) => deal.buyerId === current.session!.buyerId
            )
            .sort(
              (a, b) =>
                +new Date(b.updatedAt) -
                +new Date(a.updatedAt)
            )
        );
      }
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to load result"
      );
    }
  }, [dealId]);

  useEffect(() => {
    void load();
  }, [load]);

  const finalPrice =
    session?.acceptedPrice ??
    session?.currentPrice ??
    session?.initialPrice ??
    0;

  const startingPrice = session?.initialPrice ?? 0;

  const savings = Math.max(
    0,
    startingPrice - finalPrice
  );

  const savingsPercent =
    startingPrice > 0
      ? (savings / startingPrice) * 100
      : 0;

  const buyerOffers = useMemo(
    () =>
      offers.filter(
        (offer) => offer.actor === "BUYER_AGENT"
      ),
    [offers]
  );

  const merchantOffers = useMemo(
    () =>
      offers.filter(
        (offer) => offer.actor === "MERCHANT_AGENT"
      ),
    [offers]
  );

  const trajectory = useMemo(() => {
    if (!session) return [];

    return [
      {
        label: "START",
        price: session.initialPrice,
        round: 0,
        actor: "START",
      },
      ...offers.map((offer) => ({
        label:
          offer.actor === "BUYER_AGENT"
            ? "BUYER"
            : "MERCHANT",
        price: offer.price,
        round: offer.round,
        actor: offer.actor,
      })),
    ];
  }, [offers, session]);

  const trajectoryBounds = useMemo(() => {
    const prices = trajectory.map(
      (point) => point.price
    );

    const allPrices = [
      ...prices,
      startingPrice,
      finalPrice,
    ];

    return {
      min: Math.min(...allPrices, 0),
      max: Math.max(...allPrices, 1),
    };
  }, [trajectory, startingPrice, finalPrice]);

  if (!session) {
    return (
      <main className="dealer-result min-h-screen bg-[#05060A] text-white">
        <div className="relative flex min-h-screen items-center justify-center overflow-hidden px-6">
          <div className="result-loading-orb result-loading-orb-a" />
          <div className="result-loading-orb result-loading-orb-b" />

          <div className="relative z-10 text-center">
            <div className="result-loader mx-auto">
              <div className="result-loader-core" />
            </div>

            <p className="dealer-kicker mt-6 text-[#60A5FA]">
              LOADING FINAL DEAL
            </p>

            <p className="mt-3 text-sm text-[#52525B]">
              Reading persisted negotiation state...
            </p>

            {error && (
              <div className="mx-auto mt-5 max-w-md rounded-xl border border-[#F87171]/15 bg-[#F87171]/[0.04] px-4 py-3">
                <p className="text-xs leading-5 text-[#F87171]">
                  {error}
                </p>
              </div>
            )}
          </div>
        </div>

        <ResultStyles />
      </main>
    );
  }

  const terminal =
    session.state === "COMPLETED" ||
    session.state === "PAID";

  return (
    <main className="dealer-result min-h-screen overflow-hidden bg-[#05060A] text-white">
      <div className="result-grid" />

      <div className="result-orb result-orb-a" />
      <div className="result-orb result-orb-b" />

      <header className="relative z-20 border-b border-white/[0.07] bg-[#05060A]/80 backdrop-blur-2xl">
        <div className="mx-auto flex h-[72px] max-w-[1500px] items-center justify-between px-5 lg:px-8">
          <div className="flex min-w-0 items-center gap-4">
            <Link
              href={`/deals/${encodeURIComponent(dealId)}`}
              className="result-icon-button"
              aria-label="Back to deal"
            >
              <ArrowLeft size={15} />
            </Link>

            <div className="min-w-0">
              <div className="flex items-center gap-2">
                <p className="dealer-kicker text-[#60A5FA]">
                  DEALER / RESULT
                </p>

                {terminal && (
                  <span className="result-live-dot" />
                )}
              </div>

              <p className="dealer-mono mt-1 truncate text-[9px] text-[#52525B]">
                {session.id}
              </p>
            </div>
          </div>

          <Link
            href="/deals/create"
            className="dealer-result-button hidden items-center gap-2 rounded-xl px-4 py-3 text-[9px] font-black tracking-[0.12em] sm:flex"
          >
            CREATE ANOTHER DEAL
            <ArrowRight size={13} />
          </Link>
        </div>
      </header>

      <section className="relative z-10 mx-auto max-w-[1500px] px-5 pb-20 pt-10 lg:px-8 lg:pt-14">
        {/* HERO */}
        <div className="grid gap-6 lg:grid-cols-[1fr_0.72fr] lg:items-stretch">
          <section className="result-hero-panel relative overflow-hidden rounded-[32px] p-7 sm:p-10">
            <div className="result-hero-scan" />

            <div className="relative z-10">
              <div className="flex flex-wrap items-center gap-2">
                <span className="result-status-pill">
                  <span className="result-status-dot" />
                  {session.state}
                </span>

                <span className="result-round-pill">
                  ROUND {session.currentRound}
                </span>

                {offers.length > 0 && (
                  <span className="result-round-pill">
                    {offers.length} OFFERS
                  </span>
                )}
              </div>

              <p className="dealer-kicker mt-10 text-[#8B5CF6]">
                NEGOTIATION COMPLETE
              </p>

              <h1 className="mt-4 max-w-4xl text-5xl font-semibold leading-[0.94] tracking-[-0.06em] sm:text-7xl lg:text-[82px]">
                The deal
                <br />
                <span className="result-gradient">
                  was negotiated.
                </span>
              </h1>

              <p className="mt-7 max-w-xl text-sm leading-7 text-[#8A8F9B]">
                DEALER recorded the negotiation, validated the
                offers, enforced the business policy, and persisted
                the final authoritative value.
              </p>

              <div className="mt-9 flex flex-wrap items-center gap-3">
                <div className="result-architecture-chip">
                  <ShieldCheck
                    size={13}
                    className="text-[#22C55E]"
                  />
                  DETERMINISTIC CONTROL
                </div>

                <div className="result-architecture-chip">
                  <Sparkles
                    size={13}
                    className="text-[#A78BFA]"
                  />
                  AI-ASSISTED NEGOTIATION
                </div>
              </div>
            </div>
          </section>

          {/* FINAL VALUE */}
          <section className="result-final-card relative overflow-hidden rounded-[32px] p-7 sm:p-9">
            <div className="result-final-glow" />

            <div className="relative z-10 flex h-full flex-col justify-between">
              <div>
                <div className="flex items-center justify-between">
                  <p className="dealer-kicker text-[#60A5FA]">
                    FINAL AUTHORITATIVE VALUE
                  </p>

                  <div className="result-check">
                    <CheckCircle2 size={15} />
                  </div>
                </div>

                <p className="dealer-mono mt-5 text-5xl font-semibold tracking-[-0.06em] text-white sm:text-6xl lg:text-[68px]">
                  {money(finalPrice)}
                </p>

                <div className="mt-4 h-px bg-white/[0.07]" />

                <div className="mt-5 grid grid-cols-2 gap-5">
                  <div>
                    <p className="dealer-kicker text-[#52525B]">
                      STARTING PRICE
                    </p>

                    <p className="dealer-mono mt-2 text-sm text-[#71717A]">
                      {money(startingPrice)}
                    </p>
                  </div>

                  <div>
                    <p className="dealer-kicker text-[#22C55E]">
                      BUYER SAVING
                    </p>

                    <p className="dealer-mono mt-2 text-lg font-semibold text-[#22C55E]">
                      {money(savings)}
                    </p>
                  </div>
                </div>
              </div>

              <div className="mt-10 rounded-2xl border border-[#22C55E]/10 bg-[#22C55E]/[0.035] p-4">
                <div className="flex items-center justify-between">
                  <span className="dealer-kicker text-[#52525B]">
                    EFFECTIVE DISCOUNT
                  </span>

                  <span className="dealer-mono text-xl font-semibold text-[#22C55E]">
                    {savingsPercent.toFixed(1)}%
                  </span>
                </div>

                <div className="mt-3 h-1 overflow-hidden rounded-full bg-white/[0.05]">
                  <div
                    className="h-full rounded-full bg-gradient-to-r from-[#22C55E] to-[#60A5FA]"
                    style={{
                      width: `${Math.min(
                        100,
                        Math.max(0, savingsPercent)
                      )}%`,
                    }}
                  />
                </div>
              </div>
            </div>
          </section>
        </div>

        {/* METRICS */}
        <section className="mt-5 grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
          <Metric
            icon={<CircleDollarSign size={15} />}
            label="STARTING PRICE"
            value={money(startingPrice)}
          />

          <Metric
            icon={<Trophy size={15} />}
            label="FINAL PRICE"
            value={money(finalPrice)}
            emphasis
          />

          <Metric
            icon={<TrendingDown size={15} />}
            label="TOTAL SAVING"
            value={`${money(savings)} Â· ${savingsPercent.toFixed(1)}%`}
            success={savings > 0}
          />

          <Metric
            icon={<BarChart3 size={15} />}
            label="OFFERS / ROUNDS"
            value={`${offers.length} / ${session.currentRound}`}
          />
        </section>

        {/* TRAJECTORY + AUDIT */}
        <section className="mt-5 grid gap-5 lg:grid-cols-[1.35fr_0.65fr]">
          <Trajectory
            trajectory={trajectory}
            min={trajectoryBounds.min}
            max={trajectoryBounds.max}
          />

          <section className="dealer-panel rounded-[28px] p-5 sm:p-6">
            <div className="flex items-start justify-between gap-4">
              <div>
                <p className="dealer-kicker text-[#A78BFA]">
                  DEAL AUDIT
                </p>

                <h2 className="mt-2 text-lg font-semibold">
                  Control layer
                </h2>
              </div>

              <div className="result-audit-icon">
                <ShieldCheck size={16} />
              </div>
            </div>

            <div className="mt-7 space-y-4">
              <AuditStat
                label="Final deal state"
                value={session.state}
              />

              <AuditStat
                label="Buyer agent offers"
                value={String(buyerOffers.length)}
              />

              <AuditStat
                label="Merchant agent offers"
                value={String(merchantOffers.length)}
              />

              <AuditStat
                label="Control events"
                value={String(audit.length)}
              />

              <AuditStat
                label="Maximum rounds"
                value={String(session.maxRounds)}
              />
            </div>

            <div className="mt-7 rounded-2xl border border-[#60A5FA]/10 bg-[#60A5FA]/[0.025] p-4">
              <div className="flex items-start gap-3">
                <ShieldCheck
                  size={15}
                  className="mt-0.5 shrink-0 text-[#60A5FA]"
                />

                <div>
                  <p className="dealer-kicker text-[#60A5FA]">
                    GOVERNANCE CONFIRMED
                  </p>

                  <p className="mt-2 text-[10px] leading-5 text-[#71717A]">
                    AI proposals were processed through DEALER's
                    deterministic control layer. AI did not directly
                    execute payment.
                  </p>
                </div>
              </div>
            </div>
          </section>
        </section>

        {/* AGENT SUMMARY */}
        <section className="mt-5 grid gap-5 md:grid-cols-2">
          <AgentSummary
            type="BUYER"
            count={buyerOffers.length}
            latest={
              buyerOffers[buyerOffers.length - 1]?.price
            }
          />

          <AgentSummary
            type="MERCHANT"
            count={merchantOffers.length}
            latest={
              merchantOffers[merchantOffers.length - 1]?.price
            }
          />
        </section>

        {/* HISTORY */}
        <section className="dealer-panel mt-5 overflow-hidden rounded-[28px]">
          <div className="flex items-center justify-between border-b border-white/[0.06] px-5 py-5 sm:px-6">
            <div>
              <p className="dealer-kicker text-[#60A5FA]">
                DEAL HISTORY
              </p>

              <h2 className="mt-2 text-lg font-semibold">
                Previous negotiations
              </h2>
            </div>

            <span className="dealer-mono text-[9px] text-[#52525B]">
              {history.length} DEALS
            </span>
          </div>

          {history.length === 0 ? (
            <div className="px-6 py-10 text-center text-xs text-[#52525B]">
              No previous negotiations recorded.
            </div>
          ) : (
            <div className="divide-y divide-white/[0.045]">
              {history.slice(0, 8).map((deal) => {
                const dealFinal =
                  deal.acceptedPrice ??
                  deal.currentPrice ??
                  deal.initialPrice;

                const dealSaving = Math.max(
                  0,
                  deal.initialPrice - dealFinal
                );

                const isCurrent =
                  deal.id === session.id;

                return (
                  <Link
                    key={deal.id}
                    href={
                      isCurrent
                        ? `/deals/${encodeURIComponent(
                            deal.id
                          )}/result`
                        : `/deals/${encodeURIComponent(
                            deal.id
                          )}`
                    }
                    className={`result-history-row ${
                      isCurrent
                        ? "bg-white/[0.025]"
                        : ""
                    }`}
                  >
                    <div className="flex min-w-0 items-center gap-3">
                      <div className="result-history-icon">
                        {isCurrent ? (
                          <Trophy
                            size={13}
                            className="text-[#60A5FA]"
                          />
                        ) : (
                          <Clock3
                            size={13}
                            className="text-[#52525B]"
                          />
                        )}
                      </div>

                      <div className="min-w-0">
                        <div className="flex items-center gap-2">
                          <p className="dealer-kicker text-[#71717A]">
                            {deal.state}
                          </p>

                          {isCurrent && (
                            <span className="dealer-kicker text-[#60A5FA]">
                              CURRENT
                            </span>
                          )}
                        </div>

                        <p className="dealer-mono mt-1 truncate text-[9px] text-[#52525B]">
                          {deal.id}
                        </p>
                      </div>
                    </div>

                    <div className="flex items-center gap-5 sm:gap-8">
                      <div>
                        <p className="dealer-kicker text-[#52525B]">
                          FINAL
                        </p>

                        <p className="dealer-mono mt-1 text-sm text-[#A1A1AA]">
                          {money(dealFinal)}
                        </p>
                      </div>

                      <div>
                        <p className="dealer-kicker text-[#52525B]">
                          SAVING
                        </p>

                        <p className="dealer-mono mt-1 text-sm text-[#22C55E]">
                          {money(dealSaving)}
                        </p>
                      </div>

                      <ArrowRight
                        size={14}
                        className="text-[#3F3F46]"
                      />
                    </div>
                  </Link>
                );
              })}
            </div>
          )}
        </section>

        {/* CTA */}
        <section className="mt-8 grid gap-3 sm:grid-cols-2">
          <Link
            href="/deals/create"
            className="dealer-result-button flex items-center justify-center gap-2 rounded-2xl px-5 py-5 text-[10px] font-black tracking-[0.14em]"
          >
            <Sparkles size={14} />
            START NEW NEGOTIATION
          </Link>

          <Link
            href="/"
            className="flex items-center justify-center gap-2 rounded-2xl border border-white/[0.08] bg-white/[0.025] px-5 py-5 text-[10px] font-black tracking-[0.14em] text-[#A1A1AA] transition hover:border-white/[0.14] hover:bg-white/[0.04] hover:text-white"
          >
            BACK TO DEALER
          </Link>
        </section>

        <footer className="mt-12 border-t border-white/[0.05] pt-6 text-center">
          <p className="dealer-kicker text-[#3F3F46]">
            AI PROPOSES Â· POLICY VALIDATES Â· STATE MACHINE CONTROLS Â·
            PAYMENT SERVICE EXECUTES
          </p>
        </footer>
      </section>

      <ResultStyles />
    </main>
  );
}

function Metric({
  icon,
  label,
  value,
  emphasis = false,
  success = false,
}: {
  icon: ReactNode;
  label: string;
  value: string;
  emphasis?: boolean;
  success?: boolean;
}) {
  return (
    <div
      className={`dealer-panel rounded-2xl p-4 ${
        emphasis
          ? "border-[#60A5FA]/15"
          : ""
      }`}
    >
      <div className="flex items-center gap-2 text-[#52525B]">
        {icon}

        <span className="dealer-kicker">
          {label}
        </span>
      </div>

      <p
        className={`dealer-mono mt-3 text-sm ${
          success
            ? "text-[#22C55E]"
            : emphasis
              ? "text-white"
              : "text-[#D4D4D8]"
        }`}
      >
        {value}
      </p>
    </div>
  );
}

function AuditStat({
  label,
  value,
}: {
  label: string;
  value: string;
}) {
  return (
    <div className="flex items-center justify-between border-b border-white/[0.045] pb-3 last:border-0 last:pb-0">
      <span className="text-xs text-[#71717A]">
        {label}
      </span>

      <span className="dealer-mono text-[10px] text-[#D4D4D8]">
        {value}
      </span>
    </div>
  );
}

      <div className="flex items-center justify-between">
        <div className="flex items-center gap-3">

            <span />
          </div>

            <p

function Trajectory({
  trajectory,
  min,
  max,
}: {
  trajectory: Array<{
    label: string;
    price: number;
    round: number;
    actor: string;
  }>;
  min: number;
  max: number;
}) {
  const range = Math.max(1, max - min);

  return (
    <section className="dealer-panel rounded-[28px] p-5 sm:p-6">
      <div className="flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between">
        <div>
          <p className="dealer-kicker text-[#60A5FA]">
            NEGOTIATION TRAJECTORY
          </p>

          <h2 className="mt-2 text-lg font-semibold">
            Actual offer movement
          </h2>
        </div>

        <div className="flex items-center gap-4">
          <Legend
            label="BUYER"
            className="text-[#60A5FA]"
          />

          <Legend
            label="MERCHANT"
            className="text-[#A78BFA]"
          />

          <Legend
            label="START"
            className="text-[#71717A]"
          />
        </div>
      </div>

      {trajectory.length < 2 ? (
        <div className="flex h-[300px] items-center justify-center rounded-2xl border border-white/[0.05] bg-black/20">
          <div className="text-center">
            <BarChart3
              size={20}
              className="mx-auto text-[#3F3F46]"
            />

            <p className="mt-3 text-xs text-[#52525B]">
              Not enough offers to plot a trajectory yet.
            </p>
          </div>
        </div>
      ) : (
        <div className="mt-7 overflow-x-auto">
          <div
            className="relative min-w-[620px] rounded-2xl border border-white/[0.05] bg-black/20 px-5 py-6"
            style={{ height: 330 }}
          >
            <div className="absolute inset-x-6 top-[25%] border-t border-dashed border-white/[0.045]" />
            <div className="absolute inset-x-6 top-[50%] border-t border-dashed border-white/[0.045]" />
            <div className="absolute inset-x-6 top-[75%] border-t border-dashed border-white/[0.045]" />

            <div className="absolute left-3 top-4 dealer-kicker text-[#3F3F46]">
              HIGH
            </div>

            <div className="absolute bottom-4 left-3 dealer-kicker text-[#3F3F46]">
              LOW
            </div>

            <div className="flex h-full items-end gap-3 pl-8">
              {trajectory.map((point, index) => {
                const normalized =
                  (point.price - min) / range;

                const height =
                  18 + normalized * 67;

                const isBuyer =
                  point.label === "BUYER";

                const isMerchant =
                  point.label === "MERCHANT";

                return (
                  <div
                    key={`${point.round}-${index}`}
                    className="group flex h-full min-w-[48px] flex-1 flex-col justify-end"
                  >
                    <div className="mb-2 text-center opacity-80 transition group-hover:opacity-100">
                      <p className="dealer-mono text-[8px] text-[#A1A1AA]">
                        {money(point.price)}
